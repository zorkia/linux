Collector

nano mqtt_to_sqlite.py
import json
from pathlib import Path
import sqlite3
import paho.mqtt.client as mqtt

database = Path.home() / "iot-data" / "device_status.db"
connection = sqlite3.connect(database, check_same_thread=False)
cursor = connection.cursor()

cursor.execute(
    "CREATE TABLE IF NOT EXISTS device_logs ("
    "id INTEGER PRIMARY KEY AUTOINCREMENT,"
    "created_at DATETIME DEFAULT CURRENT_TIMESTAMP,"
    "device_id TEXT NOT NULL,"
    "location TEXT NOT NULL,"
    "temperature REAL NOT NULL,"
    "humidity REAL NOT NULL,"
    "status TEXT NOT NULL,"
    "note TEXT)"
)
connection.commit()

def on_message(client, userdata, message):
    try:
        data = json.loads(message.payload.decode())
        temperature = float(data["temperature"])
        cursor.execute(
            "INSERT INTO device_logs "
            "(device_id,location,temperature,humidity,status,note) "
            "VALUES (?,?,?,?,?,?)",
            (
                data["device_id"],
                data.get("location", "Unknown"),
                temperature,
                float(data["humidity"]),
                data.get("status") or ("WARNING" if temperature > 35 else "NORMAL"),
                data.get("note", "mqtt collector")
            )
        )
        connection.commit()
        print(f"SAVED: {data}")
    except Exception as error:
        print(f"ERROR: {error}")

client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2)
client.on_message = on_message
client.connect("127.0.0.1", 1883, 60)
client.subscribe("pkru/iot/+/data")
print("Waiting for MQTT data...")
client.loop_forever()
