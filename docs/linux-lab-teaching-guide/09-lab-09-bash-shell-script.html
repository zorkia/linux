LAB 09 - Bash Shell Script

ก่อนหน้า: LAB 08 - Linux Environment Variables และ Bash Shellกลับหน้าแรกถัดไป: LAB 10 - Python บน Linux
เครื่องมือสำหรับ LAB นี้

ใช้ nano ที่ติดตั้งใน LAB 01 เพื่อเขียน Script ส่วน bash, hostname, date และ chmod มีอยู่ใน Image ดู รวมคำสั่งและการติดตั้งล่วงหน้า

วัตถุประสงค์

เขียน Shell Script เพื่อทำงาน Linux แบบอัตโนมัติ และเตรียมแนวคิดสำหรับงานดูแล Raspberry Pi/IoT Gateway

เริ่มต้น

docker start -ai rpi-lab
su - piuser
sudo apt update
sudo apt install -y nano
mkdir -p ~/scripts
cd ~/scripts

1. Script แรก

nano hello.sh

ใส่:

#!/bin/bash

echo "Hello Raspberry Pi Linux"

บันทึก แล้ว:

chmod +x hello.sh
./hello.sh

2. Variables

สร้าง:

nano device.sh
#!/bin/bash

device_id="ESP32-001"
location="LAB"

echo "Device: $device_id"
echo "Location: $location"

รัน:

chmod +x device.sh
./device.sh

3. Command Substitution

nano system-info.sh
#!/bin/bash

hostname_value=$(hostname)
date_value=$(date)

echo "Hostname: $hostname_value"
echo "Date: $date_value"

รัน:

chmod +x system-info.sh
./system-info.sh

4. Arguments

nano argument.sh
#!/bin/bash

echo "Script: $0"
echo "Device: $1"
echo "Value: $2"

รัน:

chmod +x argument.sh
./argument.sh ESP32-01 30.5

5. if

nano check-temp.sh
#!/bin/bash

temp=$1

if [ "$temp" -gt 35 ]; then
    echo "WARNING"
else
    echo "NORMAL"
fi

รัน:

chmod +x check-temp.sh
./check-temp.sh 30
./check-temp.sh 40

6. for

nano devices.sh
#!/bin/bash

for device in ESP32-01 ESP32-02 ESP32-03
do
    echo "Checking $device"
done

รัน:

chmod +x devices.sh
./devices.sh

7. while

nano counter.sh
#!/bin/bash

count=1

while [ "$count" -le 5 ]
do
    echo "Reading $count"
    count=$((count + 1))
done

8. เขียน Log

nano logger.sh
#!/bin/bash

mkdir -p ~/logs
echo "$(date) Gateway check OK" >> ~/logs/gateway.log

รันหลายครั้ง:

chmod +x logger.sh
./logger.sh
./logger.sh
cat ~/logs/gateway.log

9. Exit Code

ls /etc
echo $?

ทดลองคำสั่งผิด:

ls /directory-does-not-exist
echo $?

โดยทั่วไป 0 หมายถึงสำเร็จ และค่าที่ไม่ใช่ 0 หมายถึงเกิดเงื่อนไขผิดพลาด

แบบฝึกหัด

สร้าง iot-status.sh ที่:

  1. แสดง Date/Time
  2. แสดง Hostname
  3. แสดง User
  4. แสดง IP ของ eth0
  5. Ping 8.8.8.8 1 ครั้ง
